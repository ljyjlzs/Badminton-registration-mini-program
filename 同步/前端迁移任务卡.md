     applyFilter: function() {
       const kw = (this.data.searchKeyword || '').trim().toLowerCase();
       const type = this.data.filterType;
       const dateStart = this.data.filterDateStart;
       const dateEnd = this.data.filterDateEnd;

       const filterFn = (item) => {
         if (kw) {
           const nameMatch = item.name && item.name.toLowerCase().includes(kw);
           const locationMatch = item.location && item.location.toLowerCase().includes(kw);
           if (!nameMatch && !locationMatch) return false;
         }
         if (type && item.type !== type) return false;
         if (dateStart && item.time < dateStart) return false;
         if (dateEnd && item.time > dateEnd + ' 23:59:59') return false;
         return true;
       };

       this.setData({
         filteredCreatedActivities: this.data.myCreatedActivities.filter(filterFn),
         filteredJoinedActivities: this.data.myJoinedActivities.filter(filterFn)
       });
     }
     ```

           <button class="btn-cancel" bindtap="hideAiDialog">取消</button>

     </view>
     ```

2. `miniprogram/pages/create-activity/create-activity.js`
   - `data` 加 `showAiDialog: false`、`aiInput: ''`
   - 加方法：
     ```js
     onAiGenerate() { this.setData({ showAiDialog: true }); },
     hideAiDialog() { this.setData({ showAiDialog: false }); },
     onAiInput(e) { this.setData({ aiInput: e.detail.value }); },
     doAiGenerate() {
       callFunction('parse-activity', 'parse', { message: this.data.aiInput })
         .then(draft => {
           this.setData({
             name: draft.name || '',
             date: this.extractDate(draft.time),
             time: this.extractTime(draft.time),
             location: draft.location || '',
             activityType: draft.type || 'doubles',
             maxPlayers: draft.maxPlayers || 8,
             showAiDialog: false
           });
           if (draft.missing && draft.missing.length > 0) {
             wx.showToast({ title: '请补充：' + draft.missing.join('、'), icon: 'none' });
           }
         })
         .catch(() => {
           wx.showToast({ title: 'AI 暂时无法识别，请手动填写', icon: 'none' });
           this.setData({ showAiDialog: false });
         });
     },
     extractDate(timeStr) {
       if (!timeStr) return '';
       return timeStr.split(' ')[0] || '';
     },
     extractTime(timeStr) {
       if (!timeStr) return '';
       return timeStr.split(' ')[1] ? timeStr.split(' ')[1].substring(0, 5) : '';
     }
     ```

**验收标准**：
- 表单顶部显示"AI 帮你填"入口
- 点击后弹出输入框，输入描述后自动填充表单字段
- AI 返回的 `missing` 字段对应项保持空白待用户手动填写
- AI 不可用时 toast 提示"AI 暂时无法识别，请手动填写"
