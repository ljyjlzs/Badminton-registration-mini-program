**JS 重写**：参考 Java 版 `my-summary.js`，需实现：

```javascript
var callFunction = require('../../utils/request').callFunction;

Page({
  data: {
    loading: true,
    periodType: 'season',
    periodLabel: '',
    periodRangeText: '',
    statCards: [],
    highlights: [],
    badges: [],
    showAiNote: false,
    aiContent: '',
    aiSourceText: '',
    generatedAtText: ''
  },

  onLoad: function() { this.loadSummary(); },

  onPullDownRefresh: function() {
    var that = this;
    this.loadSummary(function() { wx.stopPullDownRefresh(); });
  },

  switchPeriod: function(e) {
    this.setData({ periodType: e.currentTarget.dataset.type });
    this.loadSummary();
  },

  loadSummary: function(done) {
    var that = this;
    that.setData({ loading: true });
    callFunction('user-summary', 'get', {
      periodType: that.data.periodType
    }).then(function(data) {
      that.setData({
        loading: false,
        periodLabel: data.periodLabel || '我的总结',
        periodRangeText: data.periodRangeText || '',
        statCards: data.statCards || [],
        highlights: data.highlights || [],
        badges: data.badges || [],
        showAiNote: !!data.aiContent,
        aiContent: data.aiContent || '',
        aiSourceText: data.aiSourceText || 'AI 生成',
        generatedAtText: data.generatedAtText || ''
      });
      if (done) done();
    }).catch(function(err) {
      console.error('加载总结失败：', err);
      that.setData({ loading: false });
      if (done) done();
    });
  },

  copyContent: function() {
    var texts = [];
    this.data.statCards.forEach(function(c) { texts.push(c.label + '：' + c.value); });
    if (this.data.aiContent) texts.push('\n' + this.data.aiContent);
    wx.setClipboardData({ data: texts.join('\n'), success: function() {
      wx.showToast({ title: '已复制', icon: 'success' });
    }});
  },

  onShareAppMessage: function() {
    return { title: this.data.periodLabel || '我的羽毛球总结' };
  }
});
```

**JS 重写**：

```javascript
var callFunction = require('../../utils/request').callFunction;

Page({
  data: {
    activityId: '',
    activityName: '比赛顺序',
    loading: true,
    saving: false,
    locked: false,
    matches: []
  },

  onLoad: function(options) {
    this.setData({
      activityId: options.activityId || '',
      activityName: options.activityName ? decodeURIComponent(options.activityName) : '比赛顺序'
    });
    if (this.data.activityId) this.loadOrder();
  },

  loadOrder: function() {
    var that = this;
    that.setData({ loading: true });
    callFunction('match-order', 'get', { activityId: that.data.activityId })
      .then(function(data) {
        var matches = (data.matches || []).map(function(m, i) {
          return {
            id: m.id,
            status: m.status || 'pending',
            statusText: m.status === 'finished' ? '已结束' : (m.status === 'playing' ? '比赛中' : '待开始'),
            roundOrder: m.roundOrder || i + 1,
            team1DisplayName: m.team1Name || '队伍一',
            team2DisplayName: m.team2Name || '队伍二',
            team1MemberNames: (m.team1MemberNames || []).join('/'),
            team2MemberNames: (m.team2MemberNames || []).join('/')
          };
        });
        that.setData({ loading: false, locked: !!data.locked, matches: matches });
      }).catch(function(err) {
        that.setData({ loading: false });
        wx.showToast({ title: err.message || '加载失败', icon: 'none' });
      });
  },

  moveUp: function(e) {
    if (this.data.locked) return;
    var index = Number(e.currentTarget.dataset.index);
    if (index <= 0) return;
    this.swap(index, index - 1);
  },

  moveDown: function(e) {
    if (this.data.locked) return;
    var index = Number(e.currentTarget.dataset.index);
    if (index >= this.data.matches.length - 1) return;
    this.swap(index, index + 1);
  },

  swap: function(from, to) {
    var matches = this.data.matches.map(function(item) { return Object.assign({}, item); });
    var target = matches[from];
    matches[from] = matches[to];
    matches[to] = target;
    matches.forEach(function(item, i) { item.roundOrder = i + 1; });
    this.setData({ matches: matches });
  },

  saveOrder: function() {
    var that = this;
    if (that.data.locked) {
      wx.showToast({ title: '比赛已锁定', icon: 'none' }); return;
    }
    if (that.data.saving || that.data.matches.length === 0) return;
    var orderText = that.data.matches.map(function(m, i) {
      return '第 ' + (i + 1) + ' 场：' + m.team1DisplayName + ' vs ' + m.team2DisplayName;
    }).join('\n');
    wx.showModal({
      title: '确认保存',
      content: '保存后比赛将按以下顺序进行：\n' + orderText + '\n确认保存？',
      success: function(res) {
        if (!res.confirm) return;
        that.setData({ saving: true });
        callFunction('match-order', 'update', {
          activityId: that.data.activityId,
          matchIds: that.data.matches.map(function(m) { return m.id; })
        }).then(function() {
          that.setData({ saving: false });
          wx.showToast({ title: '顺序已保存', icon: 'success' });
          setTimeout(function() { wx.navigateBack(); }, 800);
        }).catch(function(err) {
          that.setData({ saving: false });
          wx.showToast({ title: err.message || '保存失败', icon: 'none' });
        });
      }
    });
  }
});
```
