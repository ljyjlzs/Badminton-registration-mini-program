**JS 方法补充**：
```javascript
parseActivity: function() {
  var input = this.data.aiInput.trim();
  if (!input) return;
  this.setData({ 'aiSuggestion.show': true, 'aiSuggestion.loading': true });
  callFunction('ai-parse-activity', 'parse', { input: input }).then(function(data) {
    this.setData({
      'aiSuggestion.loading': false,
      'aiSuggestion.title': 'AI 解析结果',
      'aiSuggestion.desc': '以下是 AI 理解的活动信息，确认后预填表单',
      'aiSuggestion.draft': data
    });
  }.bind(this)).catch(function() {
    this.setData({ 'aiSuggestion.show': false });
    wx.showToast({ title: 'AI 解析失败', icon: 'none' });
  }.bind(this));
},
applyAiDraft: function() {
  var draft = this.data.aiSuggestion.draft;
  if (!draft) return;
  this.setData({
    name: draft.name || this.data.name,
    location: draft.location || this.data.location,
    date: draft.date || this.data.date,
    time: draft.time || this.data.time,
    activityType: draft.type || this.data.activityType,
    'aiSuggestion.show': false
  });
},
discardAiDraft: function() {
  this.setData({ 'aiSuggestion.show': false });
}
```

```javascript
fetchRecommend: function() {
  this.setData({ 'recommend.show': true, 'recommend.loading': true });
  callFunction('recommend-time', 'suggest', {
    activityType: this.data.activityType,
    location: this.data.location
  }).then(function(data) {
    this.setData({
      'recommend.loading': false,
      'recommend.list': data.list || [],
      'recommend.aiNote': data.aiNote || '',
      'recommend.reliable': data.reliable !== false,
      'recommend.fallbackTip': data.fallbackTip || ''
    });
  }.bind(this)).catch(function() {
    this.setData({ 'recommend.show': false });
  }.bind(this));
},
closeRecommend: function() {
  this.setData({ 'recommend.show': false });
},
applyRecommend: function(e) {
  var item = this.data.recommend.list[e.currentTarget.dataset.index];
  if (item && item.suggestedTime) {
    var parts = item.suggestedTime.split(' ');
    if (parts.length >= 2) {
      this.setData({ date: parts[0], time: parts[1] });
    }
  }
}
```

```javascript
startVoiceInput: function() {
  var that = this;
  if (!that.data.voiceConfigured) {
    wx.showToast({ title: '语音识别未配置', icon: 'none' });
    return;
  }
  that.setData({ voiceRecording: true, voiceRecognizing: false });
  // 使用微信录音管理器
  var recorderManager = wx.getRecorderManager();
  that._recorderManager = recorderManager;
  that._recorderManager.onStop(function(res) {
    if (res.tempFilePath) {
      that.setData({ voiceRecognizing: true });
      that.recognizeVoice(res.tempFilePath);
    }
  });
  that._recorderManager.onError(function() {
    that.setData({ voiceRecording: false });
    wx.showToast({ title: '录音失败', icon: 'none' });
  });
  that._recorderManager.start({
    format: 'mp3',
    sampleRate: 16000,
    numberOfChannels: 1
  });
},

stopVoiceInput: function() {
  if (this._recorderManager) {
    this._recorderManager.stop();
  }
  this.setData({ voiceRecording: false });
},

recognizeVoice: function(filePath) {
  var that = this;
  // 上传到云函数进行语音识别
  wx.cloud.uploadFile({
    cloudPath: 'voice-temp/' + Date.now() + '.mp3',
    filePath: filePath,
    success: function(uploadRes) {
      callFunction('voice-score', 'recognize', {
        fileID: uploadRes.fileID,
        activityId: that.data.activityId
      }).then(function(data) {
        that.setData({
          voiceRecognizing: false,
          voiceShowPanel: true,
          voiceResult: data.result || null
        });
      }).catch(function() {
        that.setData({ voiceRecognizing: false });
        wx.showToast({ title: '识别失败，请手动输入', icon: 'none' });
      });
    },
    fail: function() {
      that.setData({ voiceRecognizing: false });
      wx.showToast({ title: '上传失败', icon: 'none' });
    }
  });
},

cancelVoiceResult: function() {
  this.setData({ voiceShowPanel: false, voiceResult: null });
},

confirmVoiceResult: function() {
  var that = this;
  var result = that.data.voiceResult;
  if (!result || that.data.voiceSubmitting) return;
  that.setData({ voiceSubmitting: true });
  callFunction('score', 'update', {
    activityId: that.data.activityId,
    matchId: result.matchId,
    team1Score: result.team1Score,
    team2Score: result.team2Score
  }).then(function() {
    that.setData({ voiceSubmitting: false, voiceShowPanel: false, voiceResult: null });
    wx.showToast({ title: '记分成功', icon: 'success' });
    that.loadActivityDetail();
  }).catch(function() {
    that.setData({ voiceSubmitting: false });
    wx.showToast({ title: '提交失败', icon: 'none' });
  });
}
```
