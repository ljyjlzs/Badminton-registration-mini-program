```javascript
  const activityTime = new Date(activity.data.time);
  
  // 查询用户所有 active 状态的报名
  const myRegs = await db.collection('registrations')
    .where({
      userId: openid,
      applyStatus: 'active',
      cancelStatus: _.neq('approved')
    })
    .get();
  
  // 获取所有已报名的活动
  const activityIds = myRegs.data.map(r => r.activityId);
  const activities = await db.collection('activities')
    .where({ _id: _.in(activityIds) })
    .get();
  
  // 检查时间冲突（前后2小时内）
  const conflicts = [];
  for (const act of activities.data) {
    if (act._id === activityId) continue;
    
    const actTime = new Date(act.time);
    const timeDiff = Math.abs(activityTime - actTime);
    const hoursDiff = timeDiff / (1000 * 60 * 60);
    
    if (hoursDiff <= 2) {
      conflicts.push({
        activityId: act._id,
        name: act.name,
        time: act.time,
        location: act.location
      });
    }
  }
  
  return conflicts;
}
```

```javascript
// 获取比赛排序
async function getMatchOrder(activityId) {
  const matches = await db.collection('matches')
    .where({ activityId })
    .orderBy('roundOrder', 'asc')
    .get();
  
  // 查询队伍信息
  const teamIds = new Set();
  matches.data.forEach(m => {
    if (m.team1Id) teamIds.add(m.team1Id);
    if (m.team2Id) teamIds.add(m.team2Id);
  });
  
  const teams = await db.collection('teams')
    .where({ _id: _.in([...teamIds]) })
    .get();
  
  const teamsMap = {};
  teams.data.forEach(t => {
    teamsMap[t._id] = t;
  });
  
  return matches.data.map(m => ({
    matchId: m._id,
    roundOrder: m.roundOrder,
    team1Name: teamsMap[m.team1Id]?.name || '队伍1',
    team2Name: teamsMap[m.team2Id]?.name || '队伍2',
    status: m.status,
    team1Score: m.team1Score,
    team2Score: m.team2Score
  }));
}

// 保存比赛排序
async function saveMatchOrder(activityId, matchOrders, openid) {
  // 验证是否是组织者
  const activity = await db.collection('activities').doc(activityId).get();
  if (activity.data.creatorOpenid !== openid && activity.data._openid !== openid) {
    return { success: false, message: '只有组织者可以调整比赛顺序' };
  }
  
  // 批量更新
  for (const item of matchOrders) {
    await db.collection('matches').doc(item.matchId).update({
      data: { roundOrder: item.roundOrder }
    });
  }
  
  return { success: true, message: '保存成功' };
```

```javascript
// 修改比分
async function modifyScore(payload, openid) {
  const { activityId, matchId, team1Score, team2Score } = payload;
  
  // 验证比分范围
  if (team1Score < 0 || team1Score > 30 || team2Score < 0 || team2Score > 30) {
    return { success: false, message: '比分必须在0-30之间' };
  }
  if (team1Score === team2Score) {
    return { success: false, message: '不允许平局' };
  }
  
  // 查询比赛信息
  const match = await db.collection('matches').doc(matchId).get();
  
  // 查询活动信息
  const activity = await db.collection('activities').doc(activityId).get();
  
  // 验证权限
  const isOrganizer = activity.data.creatorOpenid === openid || activity.data._openid === openid;
  const team1 = await db.collection('teams').doc(match.data.team1Id).get();
  const team2 = await db.collection('teams').doc(match.data.team2Id).get();
  const isTeamMember = 
    team1.data.members.includes(openid) || 
    team2.data.members.includes(openid);
  
  if (!isOrganizer && !isTeamMember) {
    return { success: false, message: '只有对局双方或组织者可以修改比分' };
  }
  
  // 更新比赛比分
  await db.collection('matches').doc(matchId).update({
    data: {
      team1Score: team1Score,
      team2Score: team2Score,
      updatedAt: db.serverDate()
    }
  });
  
  // 重新计算积分
  // 删除旧积分记录
  await db.collection('scores').where({ matchId }).remove();
  
  // 添加新积分记录
  const scoreChange = team1Score - team2Score;
  
  for (const memberId of team1.data.members) {
    await db.collection('scores').add({
      data: {
        activityId,
        matchId,
        userId: memberId,
        source: 'group',
        scoreChange: scoreChange,
        createdAt: db.serverDate()
      }
    });
  }
  
  for (const memberId of team2.data.members) {
    await db.collection('scores').add({
      data: {
        activityId,
        matchId,
        userId: memberId,
        source: 'group',
        scoreChange: -scoreChange,
        createdAt: db.serverDate()
      }
    });
  }
  
  // 更新 rating_changes
  await db.collection('rating_changes').where({ matchId }).remove();
  
  // ... 添加新的 rating_changes 记录
  
  return { success: true, message: '比分修改成功' };
}
```

```javascript
// 活动结束时生成徽章
async function generateBadges(activityId) {
  // 查询所有比赛记录
  const matches = await db.collection('matches')
    .where({ activityId, status: 'finished' })
    .get();
  
  // 查询所有积分记录
  const scores = await db.collection('scores')
    .where({ activityId })
    .get();
  
  // 统计每个用户的总分
  const userScores = {};
  scores.data.forEach(s => {
    if (!userScores[s.userId]) {
      userScores[s.userId] = {
        totalScore: 0,
        matchCount: 0,
        winCount: 0,
        maxWinScore: 0
      };
    }
    userScores[s.userId].totalScore += s.scoreChange;
    userScores[s.userId].matchCount++;
    if (s.scoreChange > 0) userScores[s.userId].winCount++;
    if (s.scoreChange > userScores[s.userId].maxWinScore) {
      userScores[s.userId].maxWinScore = s.scoreChange;
    }
  });
  
  // 查询所有报名记录
  const registrations = await db.collection('registrations')
    .where({ activityId, applyStatus: 'active' })
    .get();
  
  const badges = [];
  
  // MVP：总分最高
  let mvpUserId = null;
  let mvpScore = -Infinity;
  for (const [userId, stats] of Object.entries(userScores)) {
    if (stats.totalScore > mvpScore) {
      mvpScore = stats.totalScore;
      mvpUserId = userId;
    }
  }
  if (mvpUserId) {
    const mvpReg = registrations.data.find(r => r.userId === mvpUserId);
    badges.push({
      activityId,
      userId: mvpUserId,
      badgeCode: 'mvp',
      title: 'MVP',
      description: `${mvpReg?.nickname || '未知'} 本次比赛得分最高`,
      createdAt: db.serverDate()
    });
  }
  
  // 努力奖：参与比赛最多
  let hardWorkerUserId = null;
  let maxMatches = 0;
  for (const [userId, stats] of Object.entries(userScores)) {
    if (stats.matchCount > maxMatches) {
      maxMatches = stats.matchCount;
      hardWorkerUserId = userId;
    }
  }
  if (hardWorkerUserId) {
    const hardWorkerReg = registrations.data.find(r => r.userId === hardWorkerUserId);
    badges.push({
      activityId,
      userId: hardWorkerUserId,
      badgeCode: 'hard_worker',
      title: '努力奖',
      description: `${hardWorkerReg?.nickname || '未知'} 完成${maxMatches}场以上比赛`,
      createdAt: db.serverDate()
    });
  }
  
  // 最大胜分：单场胜分最多
  let biggestWinUserId = null;
  let maxWinScore = 0;
  for (const [userId, stats] of Object.entries(userScores)) {
    if (stats.maxWinScore > maxWinScore) {
      maxWinScore = stats.maxWinScore;
      biggestWinUserId = userId;
    }
  }
  if (biggestWinUserId) {
    const biggestWinReg = registrations.data.find(r => r.userId === biggestWinUserId);
    badges.push({
      activityId,
      userId: biggestWinUserId,
      badgeCode: 'biggest_win',
      title: '最大胜分',
      description: `${biggestWinReg?.nickname || '未知'} 单场胜分${maxWinScore}分`,
      createdAt: db.serverDate()
    });
  }
  
  // 保存徽章
  for (const badge of badges) {
    await db.collection('badges').add({ data: badge });
  }
  
  return badges;
}
```

```javascript
async function getDetailedStats(openid, period) {
  // 计算时间范围
  const now = new Date();
  let startDate = null;
  
  if (period === 'season') {
    startDate = new Date(now);
    startDate.setMonth(startDate.getMonth() - 3);
  } else if (period === 'year') {
    startDate = new Date(now.getFullYear(), 0, 1);
  }
  
  // 查询积分记录
  let scoresQuery = db.collection('scores').where({ userId: openid });
  if (startDate) {
    scoresQuery = scoresQuery.where({ createdAt: _.gte(startDate) });
  }
  const scores = await scoresQuery.get();
  
  // 统计基础数据
  const matchSet = new Set();
  let totalScore = 0;
  let winCount = 0;
  
  scores.data.forEach(s => {
    matchSet.add(s.matchId);
    totalScore += s.scoreChange;
    if (s.scoreChange > 0) winCount++;
  });
  
  const matchCount = matchSet.size;
  const winRate = matchCount > 0 ? (winCount / matchCount * 100).toFixed(1) : 0;
  
  // 查询报名记录
  let regsQuery = db.collection('registrations')
    .where({ userId: openid, applyStatus: 'active' });
  if (startDate) {
    regsQuery = regsQuery.where({ createdAt: _.gte(startDate) });
  }
  const regs = await regsQuery.get();
  const activityCount = new Set(regs.data.map(r => r.activityId)).size;
  
  // 查询徽章
  const badges = await db.collection('badges').where({ userId: openid }).get();
  
  // 查询最近比赛
  const recentMatches = [];
  for (const matchId of [...matchSet].slice(0, 10)) {
    const match = await db.collection('matches').doc(matchId).get();
    if (match.data) {
      recentMatches.push(match.data);
    }
  }
  
  // 排名趋势
  const rankingTrend = scores.data
    .sort((a, b) => new Date(b.createdAt) - new Date(a.createdAt))
    .slice(0, 5)
    .map(s => ({
      date: s.createdAt,
      scoreChange: s.scoreChange
    }));
  
  return {
    activityCount,
    matchCount,
    winRate: parseFloat(winRate),
    totalScore,
    badges: badges.data,
    recentMatches,
    rankingTrend
  };
}
```

```javascript
    return { success: false, message: '只有报名者本人可以解绑搭档' };
  }
  
  const partnerId = reg.data.partnerId;
  
  // 解绑
  await db.collection('registrations').doc(registrationId).update({
    data: { partnerId: null }
  });
  
  // 解绑对方
  if (partnerId) {
    const partnerReg = await db.collection('registrations')
      .where({
        activityId: reg.data.activityId,
        userId: partnerId
      })
      .get();
    
    if (partnerReg.data.length > 0) {
      await db.collection('registrations').doc(partnerReg.data[0]._id).update({
        data: { partnerId: null }
      });
    }
  }
  
  return { success: true, message: '解绑成功' };
}
```

```javascript
async function updateMaxPlayers(activityId, maxPlayers, openid) {
  // 验证人数上限
  if (maxPlayers < 1) {
    return { success: false, message: '人数上限必须大于0' };
  }
  
  // 查询活动信息
  const activity = await db.collection('activities').doc(activityId).get();
  
  // 验证权限
  if (activity.data.creatorOpenid !== openid && activity.data._openid !== openid) {
    return { success: false, message: '只有组织者可以调整人数上限' };
  }
  
  // 验证：减少时不能低于当前已报名人数
  const validCount = await db.collection('registrations')
    .where({
      activityId,
      applyStatus: 'active',
      cancelStatus: _.neq('approved')
    })
    .count();
  
  if (maxPlayers < validCount.total) {
    return { success: false, message: `人数上限不能低于当前已报名人数（${validCount.total}人）` };
  }
  
  // 更新人数上限
  await db.collection('activities').doc(activityId).update({
    data: { maxPlayers }
  });
  
  return { success: true, message: '人数上限更新成功' };
}
```

#### 3.2.4 `match-reminders`（修改）

**功能**：比赛提醒

**函数签名**：
```javascript
exports.main = async (event, context) => {
  const { action, payload } = event;
  // action: 'get'
  // payload: { activityId }
```

**核心逻辑**：
```javascript
async function getReminders(activityId, openid) {
  // 查询用户的队伍
  const myTeams = await db.collection('teams')
    .where({
      activityId,
      members: openid
    })
    .get();
  
  const myTeamIds = myTeams.data.map(t => t._id);
  
  // 查询这些队伍的比赛
  const matches1 = await db.collection('matches')
    .where({
      activityId,
      status: 'pending',
      team1Id: _.in(myTeamIds)
    })
    .get();
  
  const matches2 = await db.collection('matches')
    .where({
      activityId,
      status: 'pending',
      team2Id: _.in(myTeamIds)
    })
    .get();
  
  const allMatches = [...matches1.data, ...matches2.data];
  
  // 筛选10分钟内的比赛
  const now = new Date();
  const tenMinutesLater = new Date(now.getTime() + 10 * 60 * 1000);
  
  const reminders = allMatches.filter(m => {
    if (m.scheduledTime) {
      const scheduledTime = new Date(m.scheduledTime);
      return scheduledTime <= tenMinutesLater && scheduledTime >= now;
    }
    return false;
  });
  
  return {
    hasReminder: reminders.length > 0,
    reminders: reminders.map(m => ({
      matchId: m._id,
      scheduledTime: m.scheduledTime,
      message: '您有一场比赛即将开始'
    }))
  };
}
```

## 五、错误处理

### 5.1 错误码定义

| 错误码 | 说明 | 处理方式 |
|--------|------|----------|
| `SUCCESS` | 成功 | 正常处理 |
| `PARAM_ERROR` | 参数错误 | 提示用户检查输入 |
| `NOT_FOUND` | 资源不存在 | 提示用户 |
| `FORBIDDEN` | 无权限 | 提示用户 |
| `DUPLICATE` | 重复操作 | 提示用户 |
| `FULL` | 人数已满 | 提示用户加入候补 |
| `CONFLICT` | 时间冲突 | 提示用户确认 |

### 5.2 错误处理流程

```javascript
try {
  const result = await callFunction('join-activity', 'join', payload);
  if (result.success) {
    // 成功处理
  } else {
    // 错误处理
    wx.showToast({
      title: result.message,
      icon: 'none'
    });
  }
} catch (err) {
  // 异常处理
  wx.showToast({
    title: '网络错误，请重试',
    icon: 'none'
  });
}
```
